gap> # CUBIC FOURFOLD AUTOMORPHISM SEARCH: FULL H^2(G,C3), DET FILTER + SMOOTHNESS VERSION
gap> #
gap> # Purpose
gap> # -------
gap> # Given
gap> #   * gid   = [n,k], the GAP SmallGroup ID of the proposed projective group G;
gap> #   * Kgens = generators of a fixed strict 6-dimensional matrix lift K of G_s,
gap> #             with E(3)*IdentityMat(6) contained in K;
gap> #
gap> # this program performs four steps, with an integrated determinant filter:
gap> #
gap> #   S1 := S_1(gid, Kgens);
gap> #      Prepare G and K, compute dim H^2(G,C3), the 3-part of the Schur
gap> #      multiplier, and the normal copies of K/C3 in G.
gap> #
gap> #   S2 := S_2(S1);
gap> #      Enumerate every nonzero class in H^2(G,C3), retain exactly the
gap> #      C3-central extensions whose kernel is contained in the derived subgroup
gap> #      (the C3-stem classes), find all faithful 6-dimensional representations
gap> #      containing the fixed K, discard candidates on which determinant is
gap> #      identically 1, and remove GL(6)-conjugate matrix images.
gap> #
gap> #   S3 := S_3(S2);
gap> #      Compute the strict cubic invariant space, the matrix centralizer
gap> #      dimension, and
gap> #          family dimension = dim V_3(H) - dim C_GL6(H)
gap> #      for every surviving matrix-image class.
gap> #
gap> #   S4 := S_4(S3);
gap> #      Try bounded finite-field Groebner tests on several varied members per prime.
gap> #      Every external Singular call has a hard operating-system timeout.
gap> #      If no smooth witness is found, apply quick exact sufficient tests proving
gap> #      that every member is singular; otherwise return "unknown".
gap> #
gap> # Required GAP packages
gap> # ---------------------
gap> #   cohomolo
gap> #   repsn
gap> #
gap> # Optional for Step 4 (strongly recommended)
gap> # ------------------------------------------
gap> #   a working external Singular executable, together with GNU timeout
gap> #   (or gtimeout).  No GAP singular-package call is made by Step 4.
gap> #
gap> # Mathematical convention in Step 3
gap> # ---------------------------------
gap> # Vectors of variables are row vectors and A acts by x |-> x*A.  Thus the
gap> # invariant condition is F(x*A)=F(x).
gap> #
gap> # Important
gap> # ---------
gap> # This version includes the character-table and matrix-image deduplication fixes,
gap> # and also excludes candidates for which the whole projective group is
gap> # symplectic (equivalently det is identically 1 on the strict lift).
gap> # 1. Step 2 enumerates one representative of each pair {v,-v} of nonzero
gap> #    H^2 classes.  Inverting the labelled kernel C3 sends v to -v.  Since the
gap> #    representation search tests both scalar actions E(3) and E(3)^2, the two
gap> #    classes give the same possible matrix images and strict cubic families.
gap> #    Removing this duplication is therefore complete and roughly halves the
gap> #    extension work.
gap> # 2. A nonzero H^2 class is retained in Step 2 only when its distinguished
gap> #    kernel C3 lies in [H,H].  This is the intrinsic test that its image in
gap> #    Hom(M(G),C3) is nonzero.
gap> # 3. The input K is the strict lift, not merely G_s.  This version is also
gap> #    optimized for non-liftable K: it never enumerates all elements of Aut(K),
gap> #    but only the orbit of (the natural character, the distinguished scalar C3).

>     local n, equations, A, B, row, i, j, k, idx, basis, P, t;
>
>     if Length(Bgens) <> Length(Agens) then
>         Error("The two generator lists have different lengths.");
>     fi;
>

>         return IdentityMat(6);
>     fi;
>
>     n := Length(Bgens[1]);
>     equations := [];
>
>     for t in [1 .. Length(Bgens)] do
>         B := Bgens[t];
>         A := Agens[t];
>
>         if DimensionsMat(B) <> [n, n] or DimensionsMat(A) <> [n, n] then
>             Error("Intertwiner input matrices have incompatible dimensions.");
>         fi;
>
>         for i in [1 .. n] do
>             for j in [1 .. n] do
>                 row := List([1 .. n * n], k -> 0);
>
>                 for k in [1 .. n] do
>                     idx := (k - 1) * n + j;
>                     row[idx] := row[idx] + B[i][k];
>
>                     idx := (i - 1) * n + k;
>                     row[idx] := row[idx] - A[k][j];
>                 od;
>
>                 Add(equations, row);
>             od;
>         od;
>     od;
>
>     basis := CF_NullspaceOfEquationRows(equations, n * n);

>
>     if P = fail then
>         return fail;
>     fi;

>     for t in [1 .. Length(Bgens)] do

>             Error("Internal error: the computed intertwiner does not intertwine.");

>     od;

>             Error("Every generator in Kgens must be 6 by 6.");
>         fi;
>
>         entries := Concatenation(A);
>         if not ForAll(entries, IsCyclotomic) then
>             Error("All matrix entries must be exact cyclotomic numbers.");
>         fi;
>     od;
>
>     Kmat := Group(Kgens);
>
>     if not IsFinite(Kmat) then
>         Error("The matrix group generated by Kgens is not finite.");
>     fi;
>
>     zmat := E(3) * IdentityMat(6);
>
>     if not zmat in Kmat then
>         Error(
>             "The fixed strict lift K must contain E(3)*IdentityMat(6). ",
>             "For liftable G_s use K=<linear lift of G_s,E(3)I_6>."
>         );
>     fi;

>         CF_IsScalarMatrix

>     if Length(scalarElementsK) <> 3 then

>             "A strict cubic lift cannot contain any additional scalar matrices."
>         );
>     fi;
>
>     isoK := IsomorphismPermGroup(Kmat);
>     if isoK = fail then
>         Error("Could not convert the input matrix group K to a permutation group.");
>     fi;
>
>     K := Image(isoK);
>     KId := fail;
>     if IdGroupsAvailable(Size(K)) then

>     fi;

gap> #############################################################################
gap> # 8. BOUNDED GENERIC SMOOTHNESS TESTS
gap> #############################################################################
gap>
gap> # Philosophy
gap> # ----------
gap> # This step returns exactly one of:
gap> #
gap> #   "generically smooth"
gap> #       A concrete family member has smooth reduction at a good split prime.
gap> #
gap> #   "all members singular"
gap> #       A quick exact obstruction valid for the whole invariant vector space
gap> #       was found.
gap> #
gap> #   "unknown"
gap> #       No certificate was found within the deliberately small time budget.
gap> #
gap> # Runtime control
gap> # ---------------
gap> # For each selected prime, several candidate members are tested in ONE
gap> # Singular process.  The whole batch has a hard wall-clock timeout.  Thus the
gap> # default cost is bounded by
gap> #
gap> #       maxSmoothTrials * groebnerTimeoutSeconds
gap> #
gap> # per family, up to a small process-start overhead.
gap> #
gap> # Finite-field certificate
gap> # ------------------------
gap> # Let J be the homogeneous Jacobian ideal generated by the six first
gap> # derivatives of a cubic F over GF(p), with p > 3.  If dim(J)=0, then the
gap> # affine zero set of J is only the origin: any nonzero homogeneous zero would
gap> # generate an entire affine line and force dimension at least one.  Hence F
gap> # has no projective singular point.  This gives a rigorous smoothness
gap> # certificate using only ONE Groebner basis per candidate, rather than six
gap> # separate affine-chart computations.
gap> #############################################################################
